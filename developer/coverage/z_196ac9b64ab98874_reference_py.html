1"""Image references, following the distribution reference grammar used by Docker 

2 

3See https://github.com/distribution/reference/blob/main/reference.go 

4""" 

5 

6from __future__ import annotations 

7 

8import re 

9from dataclasses import dataclass 

10 

11from .errors import InvalidReferenceError 

12 

13DOCKER_HUB = "docker.io" 

14DOCKER_HUB_API_HOST = "registry-1.docker.io" 

15# legacy names, all meaning Docker Hub 

16DOCKER_HUB_ALIASES: frozenset[str] = frozenset({"docker.io", "index.docker.io", "registry-1.docker.io"}) 

17DEFAULT_TAG = "latest" 

18 

19COMPONENT_RE = r"[a-z0-9]+(?:(?:\.|_|__|-+)[a-z0-9]+)*" 

20REPOSITORY_RE = re.compile(rf"^{COMPONENT_RE}(?:/{COMPONENT_RE})*$") 

21TAG_RE = re.compile(r"^[A-Za-z0-9_][A-Za-z0-9._-]{0,127}$") 

22DIGEST_RE = re.compile(r"^[a-z0-9]+(?:[.+_-][a-z0-9]+)*:[A-Za-z0-9=_-]{32,}$") 

23 

24 

25def is_digest(value: str | None) -> bool: 

26 return bool(value and DIGEST_RE.match(value)) 

27 

28 

29@dataclass(frozen=True) 

30class Reference: 

31 """A parsed image reference, e.g. `ghcr.io/org/app:1.2@sha256:...` 

32 

33 registry: normalized host[:port], `docker.io` for Docker Hub 

34 repository: path within the registry, with `library/` added for Docker Hub official images 

35 """ 

36 

37 registry: str 

38 repository: str 

39 tag: str | None = None 

40 digest: str | None = None 

41 

42 def __post_init__(self) -> None: 

43 if not REPOSITORY_RE.match(self.repository): 

44 raise InvalidReferenceError(f"Invalid repository name: {self.repository}") 

45 if self.tag is not None and not TAG_RE.match(self.tag): 

46 raise InvalidReferenceError(f"Invalid tag: {self.tag}") 

47 if self.digest is not None and not is_digest(self.digest): 

48 raise InvalidReferenceError(f"Invalid digest: {self.digest}") 

49 

50 @classmethod 

51 def parse(cls, ref: str) -> Reference: 

52 if not ref or ref != ref.strip(): 

53 raise InvalidReferenceError(f"Invalid reference: {ref!r}") 

54 remainder: str = ref 

55 digest: str | None = None 

56 if "@" in remainder: 

57 remainder, digest = remainder.split("@", 1) 

58 

59 registry: str = DOCKER_HUB 

60 first, sep, rest = remainder.partition("/") 

61 # a first component is a registry if it looks like a host, as Docker decides it 

62 if sep and ("." in first or ":" in first or first == "localhost" or first != first.lower()): 

63 registry = first 

64 remainder = rest 

65 

66 tag: str | None = None 

67 # a colon after the last slash is a tag, earlier ones belong to the registry port 

68 last_slash: int = remainder.rfind("/") 

69 colon: int = remainder.rfind(":") 

70 if colon > last_slash: 

71 remainder, tag = remainder[:colon], remainder[colon + 1 :] 

72 

73 registry = normalize_registry(registry) 

74 if registry == DOCKER_HUB and "/" not in remainder: 

75 remainder = f"library/{remainder}" 

76 return cls(registry=registry, repository=remainder, tag=tag, digest=digest) 

77 

78 @property 

79 def api_host(self) -> str: 

80 """Host serving the distribution API, which differs from the registry name for Docker Hub""" 

81 return DOCKER_HUB_API_HOST if self.registry == DOCKER_HUB else self.registry 

82 

83 @property 

84 def identifier(self) -> str: 

85 """What to fetch the manifest by, digest taking precedence over tag""" 

86 return self.digest or self.tag or DEFAULT_TAG 

87 

88 def with_digest(self, digest: str) -> Reference: 

89 return Reference(self.registry, self.repository, self.tag, digest) 

90 

91 def __str__(self) -> str: 

92 result: str = f"{self.registry}/{self.repository}" 

93 if self.tag: 

94 result += f":{self.tag}" 

95 if self.digest: 

96 result += f"@{self.digest}" 

97 return result 

98 

99 

100def normalize_registry(registry: str) -> str: 

101 registry = registry.lower() 

102 return DOCKER_HUB if registry in DOCKER_HUB_ALIASES else registry
